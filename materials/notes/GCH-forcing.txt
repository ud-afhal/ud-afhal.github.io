\section{復習}
$\be{\kappa}{regular}$とする．
\begin{definition}
$\kappa$のCohen realを一つ付け加える強制半順序$\mathrm{add}(\kappa,1)$は以下で定義される：
\[
\mathrm{add}(\kappa,1)=\langle 2^{<\kappa},\supseteq,\varnothing\rangle
\]
\end{definition}
\begin{remark}よく知られるように，以下が成り立つ：
\[
\mathrm{add}(\kappa,1)\forces\text{``$\bigcup\dot{G}:\kappa\to 2$''}
\]
$\bigcup G$を指して$\kappa$のCohen realという．
\end{remark}
\begin{proposition}\label{th:property}
$\mathrm{add}(\kappa,1)$は次の性質を持つ：
\begin{enumerate}
\item $<\kappa$-closed
\item $(2^{<\kappa})^+$-c.c.
\item $|\mathrm{add}(\kappa,1)|=2^{<\kappa}$
\end{enumerate}
\end{proposition}
\begin{proof}
いずれも定義から簡単に従う．
\end{proof}
\begin{proposition}
$\mathrm{add}(\kappa,1)\forces\text{``$2^{<\kappa}=\kappa$''}$．
\end{proposition}
\begin{proof}
$\alpha<\kappa$を任意に取って固定する．
\begin{phase}
\item 命題\ref{th:property}の1より
\[
\mathrm{add}(\kappa,1)\forces\text{``$\check{V}^{<\kappa}\subseteq \check{V}$''}
\]
であるから，$\mathrm{add}(\kappa,1)\forces\text{``${}^\alpha2=({}^\alpha 2)^{\check{V}}$''}$となる．

\item $2\cdot\kappa=\kappa$に注意して，以降では$\bigcup G:\kappa\to 2$を，$\langle \bigcup G(2\cdot\xi) \bigcup G(2\cdot\xi+1):\xi<\kappa\rangle$によって$\set{\texttt{00},\texttt{01},\texttt{10},\texttt{11}}$に値を取る長さ$\kappa$の列と自然に同一視する．
$p\in\mathrm{add}(\kappa,1)$についても，終端を適当に処理して同様に同一視する．

$x\subseteq\alpha$を$\lceil x\rceil=\langle \chi_x(\xi)\chi_x(\xi):\xi<\alpha\rangle$によってコーディングする．$\lceil x\rceil$は$\texttt{00},\texttt{11}$のいずれかに値を取るのに注意する．

列の中に出現した際に終端を判別するため，$\lceil x\rceil$の前後に$\texttt{01}$を付け加えた列$\texttt{01}^\frown\lceil x\rceil^\frown\texttt{01}$を考える．
\begin{enumerate}
\item[$\checkmark$] $x\in{}^\alpha 2$に対して，次で定まる$D_x$は$\mathrm{add}(\kappa,1)$の稠密部分集合：
\[
D_x=\set{p\in \mathrm{add}(\kappa,1):\text{The pattern $\texttt{01}^\frown\lceil x\rceil^\frown\texttt{01}$ appears in $p$}}
\]
\begin{reason}
任意の$p\in \mathrm{add}(\kappa,1)$に対して，必要なら$0$を付け加えて列の長さが偶数になるよう延長したのち，$\texttt{01}^\frown\lceil x\rceil^\frown\texttt{01}$ を後ろに付け加えて列$q$を作れば，$p\geq q\in D_x$．ここで，$q\in \mathrm{add}(\kappa,1)$は，任意の$\beta<\kappa$に対して$\beta+1+2+\alpha+2<\kappa$が成り立つことから従う．
\end{reason}
\end{enumerate}
故に，
\[
\mathrm{add}(\kappa,1)\forces\text{``$D_x\cap\dot{G}\neq\varnothing$''}
\]
より
\[
\mathrm{add}(\kappa,1)\forces \text{``The pattern $\texttt{01}^\frown\lceil x\rceil^\frown\texttt{01}$ appears in $\bigcup\dot{G}:\kappa\to\set{\texttt{00},\texttt{01},\texttt{10},\texttt{11}}$''}
\]

以上より，$1^\circ$に注意して
\[
\mathrm{add}(\kappa,1)\forces \text{``$\forall x\in {}^\alpha 2($ The pattern $\texttt{01}^\frown\lceil x\rceil^\frown\texttt{01}$ appears in $\bigcup\dot{G}:\kappa\to \set{\texttt{00},\texttt{01},\texttt{10},\texttt{11}})$''}
\]
特に$\mathrm{add}(\kappa,1)\forces \text{``$2^\alpha\leq\kappa$''}$より，主張が従う．
\end{phase}
\end{proof}
したがって，$\mathrm{add}(\kappa,1)$は，$\kappa$以上$(2^{<\kappa})^+$未満のところを$\kappa$に潰し，その外の共終数や基数，冪演算を保つ強制半順序である．

特に，$\mathrm{add}(\lambda^+,1)$は，$\lambda^+$以上$(2^{\lambda})^+$未満のところを$\lambda^+$に潰し，その外の共終数や基数，冪演算を保つ強制半順序である．

\section{途中までの$\mathrm{GCH}$の強制}
\begin{definition}[The canonical forcing of the GCH]\label{def:GCH-iteration}
Easton support iteration $\langle\mathbb{P}_\alpha:\alpha\in\mathrm{Ord}\rangle$を次で定める：
$\mathbb{P}_\alpha\forces\text{``$\be{\alpha}{cardinal}$''}$のとき，$\mathbb{P}_\alpha\forces\text{``$\dot{\mathbb{Q}}_\alpha=\mathrm{add}(\alpha^+,1)$''}$について$\mathbb{P}_{\alpha+1}=\mathbb{P}_{\alpha}*\dot{\mathbb{Q}}_\alpha$．それ以外のときtrival forcingを合成する．
\end{definition}
\begin{remark}
ここでEaston support iterationとは，regular cardinalのlimit stagesでdirect limit，それ以外のlimit stagesでinverse limitを取ることをいう．
\end{remark}
以降の主張は，Easton supportの代わりにfull supportを用いても成り立ち，証明はより簡潔となる．
\begin{lemma}\label{th:set}
$\langle\mathbb{P}_\alpha:\alpha\in\mathrm{Ord}\rangle$を定義\ref{def:GCH-iteration}のものとする．
$\langle\lambda_\alpha:\alpha\in\mathrm{Ord}\rangle$を再帰的に以下で定義されるものとする：
\[
\lambda_0=\omega,\quad\lambda_{\alpha+1}=(\lambda_\alpha^+)^{V^{\mathbb{P}_{\lambda_\alpha}}},\quad
\lambda_\gamma=\sup_{\alpha<\gamma}\lambda_\alpha\quad\text{for $\be{\gamma}{limit}$}
\]

このとき，任意の$\alpha$に対して以下が成立：
\begin{enumerate}
\item $\forall\xi<\alpha(\mathbb{P}_{\lambda_\alpha}\forces\text{``$2^{\lambda_\xi}=\lambda_\xi^+=\lambda_{\xi+1}$''})$

\item $\mathbb{P}_{\lambda_\alpha}\forces\text{``$\be{\lambda_\alpha}{cardinal}$''}$

\item $\mathbb{P}_{\lambda_{\alpha+1}}\forces\text{``$\check{V}[\dot{G}_{\lambda_\alpha}]^{\lambda_\alpha}\subseteq \check{V}[\dot{G}_{\lambda_\alpha}]\land 2^{\lambda_\alpha}=\lambda_{\alpha+1}=\lambda_\alpha^+$''}$

\item $\lambda_{\alpha+2}=((2^{\lambda_\alpha})^+)^{V^{\mathbb{P}_{\lambda_\alpha}}}>2^{\lambda_\alpha}$
\end{enumerate}
\end{lemma}
\begin{remark}
$\langle\lambda_\alpha:\alpha\in\mathrm{Ord}\rangle$がwell-definedであるのは，各$\mathbb{P}_{\lambda_\alpha}$がweakly homogeneousより，zero-one法則が成り立つことによる．（詳しくは\cite[Theorem IV.4.15.]{Kun11}を参照．）
\end{remark}
\begin{proof}
\begin{itemize}
\item[$\checkmark$]$\mathbb{P}_{\lambda_{\alpha+1}}\equiv \mathbb{P}_{\lambda_\alpha+1}$
\begin{reason}
 $\lambda_{\alpha+1}$の定義より，$\forall\xi\in[\lambda_\alpha+1,\lambda_{\alpha+1})(\mathbb{P}_{\xi}\forces\text{``$\neg(\be{\xi}{cardinal})$''})$より$\forall\xi\in[\lambda_\alpha+1,\lambda_{\alpha+1})(\mathbb{P}_{\xi}\forces\text{``$\be{\dot{\mathbb{Q}}_\xi}{trivial}$''})$であるから，$\mathbb{P}_{\lambda_{\alpha+1}}\equiv \mathbb{P}_{\lambda_\alpha+1}$．
\end{reason}
\item[$\checkmark$] $\alpha$ごとに，1$\implies$2,3,4が成り立つ
\begin{reason}
1を仮定する．2は明らか．2より，定義から$\mathbb{P}_{\lambda_\alpha}\forces\text{``$\be{\dot{\mathbb{Q}}_{\lambda_\alpha}=\mathrm{add}(\lambda_\alpha^+,1)}{$<\lambda_{\alpha}^+$-closed}$''}$より
\[
\mathbb{P}_{\lambda_\alpha+1}\forces\text{``$\check{V}[\dot{G}_{\lambda_\alpha}]^{\lambda_\alpha}\subseteq \check{V}[\dot{G}_{\lambda_\alpha}]\land2^{\lambda_\alpha}=\lambda_{\alpha+1}=\lambda_\alpha^+$''}
\]
故に，1${}^{\circ}$より3が従う．再び2より，$\mathbb{P}_{\lambda_\alpha}\forces\text{``$\be{\dot{\mathbb{Q}}_{\lambda_\alpha}=\mathrm{add}(\lambda_\alpha^+,1)}{$(2^{\lambda_\alpha})^+$-c.c.}$''}$より
\[
\mathbb{P}_{\lambda_\alpha+1}\forces\text{``$|(2^{\lambda_\alpha})^{\check{V}[\dot{G}_{\lambda_\alpha}]}|=\lambda_{\alpha+1}=\lambda_{\alpha}^+\land ((2^{\lambda_\alpha})^+)^{\check{V}[\dot{G}_{\lambda_\alpha}]} =\lambda_{\alpha}^{++}=\lambda_{\alpha+1}^+$''}
\]
故に，1${}^{\circ}$より$\lambda_{\alpha+2}=((2^{\lambda_\alpha})^+)^{V^{\mathbb{P}_{\lambda_\alpha}}}$．最後に，$((2^{\lambda_\alpha})^+)^{V^{\mathbb{P}_{\lambda_\alpha}}}>2^{\lambda_\alpha}$であるのは，そうでないとすると
\[
\mathbb{P}_{\lambda_\alpha}\forces\text{``$\exists f:\power(\lambda_\alpha)\supseteq\power^{\check{V}}(\lambda_\alpha)\surj (2^{\lambda_\alpha})^+$''}
\]
となって矛盾する：
\end{reason}
\end{itemize}
帰納法によって任意の$\alpha$で1が成り立つのを示す．$\alpha=0$で成り立つのは明らか．$\alpha$で1,2,3,4が成り立つなら，明らかに$\alpha+1$で1が成り立つ．$\be{\alpha}{limit}$未満での1,2,3,4の成立を仮定して，$\alpha$での1の成立を示す：

\begin{itemize}
\item[$\checkmark$]$K:=\set{\xi\leq\lambda_\alpha:\be{\mathbb{P}_\xi}{(proper) direct limit of $\langle\mathbb{P}_\zeta:\zeta<\xi\rangle$}}$に対して
\[
K\subseteq\set{\xi\leq\lambda_\alpha:\be{\xi}{inaccessible}}
\]
\begin{reason}
$\langle\mathbb{P}_\zeta:\zeta\leq\xi\rangle$はEaston support iterationより，$\xi\in K$ならば$\be{\xi}{regular}$．このときさらに，$\forall\alpha\in\mathrm{Ord}(\mathbb{P}_{\lambda_{\alpha+1}}\equiv\mathbb{P}_{\lambda_\alpha+1})$に注意して，ある$\be{\beta\leq\xi}{limit}$に対して$\xi=\lambda_\beta$と書ける．このとき，帰納法の仮定より$\gamma<\beta\leq\alpha$に対して$2^{\lambda_\gamma}<\lambda_{\gamma+2}<\lambda_{\beta}$であるから，$\be{\xi=\lambda_\beta}{strong limit}$．以上よりこのとき$\be{\xi}{inaccessible}$．
\end{reason}
\item[$\checkmark$] $\be{\xi\leq\lambda_\alpha}{inaccessible}$に対して$\forall \eta<\xi(\lambda_\eta,|\mathbb{P}_{\lambda_\eta}|<\xi)$．特に$\lambda_\xi=\xi$．
\begin{reason}
$\eta$に関する帰納法によって示す．$\eta=0$の時明らか．また，$\eta$で成り立つとすると，
\[
\mathbb{P}_{\lambda_\eta}\forces\text{``$|\dot{\mathbb{Q}}_{\lambda_\eta}|=|\mathrm{add}(\lambda_\eta^+,1)|\leq 2^{\lambda_\eta}$''}
\]
であるから，帰納法の仮定から$|\mathbb{P}_{\lambda_\eta}|<\xi$かつ$\lambda_{\eta+1}\leq(2^{\lambda_\eta})^{V^{\mathbb{P}_{\lambda_\eta}}}\leq (|\mathbb{P}_{\lambda_\eta}|^{|\mathbb{P}_{\lambda_\eta}|})^{\lambda_\eta}<\xi$であり，しがって$|\mathbb{P}_{\lambda_{\eta+1}}|= |\mathbb{P}_{\lambda_\eta+1}|<\xi$．故にこのとき$\eta+1$でも成り立つ．

$\be{\eta<\xi}{limit}$未満で成り立つとすると，$|\mathbb{P}_{\lambda_\eta}|\leq (\sup_{\zeta<\eta}|\mathbb{P}_{\lambda_\zeta}|)^{\eta}$より，帰納法の仮定から$|\mathbb{P}_{\lambda_\eta}|<\xi$．同じく帰納法の仮定より，$\lambda_\eta$の定義から$\lambda_\eta<\xi$．故にこのとき$\eta$でも成り立つ．
\end{reason}
\end{itemize}

$\eta<\alpha$を任意に取る．$K_{\lambda_{\eta}}:=K\cap(\lambda_{\eta},\lambda_\alpha]$とおく．
\[
\be{X}{$K_{\lambda_\eta}$-thin}:\iff X\subseteq [\lambda_{\eta},\lambda_\alpha)\land\forall \xi\in K_{\lambda_\eta}(\sup X\cap\xi<\xi)
\]
とすると，このとき$K_{\lambda_\eta}=\varnothing\lor|\mathbb{P}_{\lambda_{\eta}}|<\min K_{\lambda_{\eta}}$に注意して
\[
\mathbb{P}_{\lambda_\eta}\forces\text{``$\forall \be{X}{$K_{\lambda_{\eta}}$-thin}\exists\be{Y\in\check{V}}{$K_{\lambda_{\eta}}$-thin}(X\subseteq Y)$''}
\]
さらに，$\forall \zeta\in[\eta,\alpha)(\mathbb{P}_{\lambda_\zeta}\forces\text{``$\be{\dot{\mathbb{Q}}_{\lambda_\zeta}}{$<\lambda_{\eta+1}$-closed}$''})$．したがって，\cite[Theorem 5.4.]{Bau83}によって，$\mathbb{P}_{\lambda_\alpha}=\mathbb{P}_{\lambda_\eta}*\dot{\mathbb{P}}_{\lambda_\eta,\lambda_\alpha}$について$\mathbb{P}_{\lambda_\eta}\forces\text{``$\be{\dot{\mathbb{P}}_{\lambda_\eta,\lambda_\alpha}}{$<\lambda_{\eta+1}$-closed}$''}$となる．故に，$
\mathbb{P}_{\lambda_\alpha}\forces\text{``$\check{V}[\dot{G}_{\lambda_\eta}]^{<\lambda_{\eta+1}}\subseteq \check{V}[\dot{G}_{\lambda_\eta}]$''}$であるから，帰納法の仮定より$
\forall\xi<\eta(\mathbb{P}_{\lambda_\alpha}\forces\text{``$2^{\lambda_\xi}=\lambda_\xi^+=\lambda_{\xi+1}$''})$．

以上より，$\forall\xi<\alpha(\mathbb{P}_{\lambda_\alpha}\forces\text{``$2^{\lambda_\xi}=\lambda_\xi^+=\lambda_{\xi+1}$''})$となって，$\alpha$でも1が成り立つ．
\end{proof}

\begin{corollary}
$\langle\mathbb{P}_\alpha:\alpha\in\mathrm{Ord}\rangle$を定義\ref{def:GCH-iteration}のものとする．任意の$\alpha$に対して，
$\mathbb{P}_{\lambda_\alpha}$は$\lambda_\alpha$未満の基数に対する$\mathrm{GCH}$を強制する．
\end{corollary}

\section{$\mathrm{GCH}$の強制}
\begin{theorem}[folklore?]
$\langle\mathbb{P}_\alpha:\alpha\in\mathrm{Ord}\rangle$を定義\ref{def:GCH-iteration}のものとする．
class forcing poset $\mathbb{P}=\bigcup_{\alpha\in \mathrm{Ord}}\mathbb{P}_\alpha$は$\ZFC+\mathrm{GCH}$を強制する．
\end{theorem}
\begin{proof}
クラス強制法であることに由来する幾つかの確認を除いて，本質的に補題\ref{th:set}と同様の証明で示せる．クラス強制法の取り扱いについては，資料を別途作成予定．
\end{proof}
